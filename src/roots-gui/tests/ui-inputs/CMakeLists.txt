cmake_minimum_required(VERSION 3.20)
project(roots_gui_ui_inputs LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT ROOTS_GUI_UI_SOURCE OR NOT IS_DIRECTORY "${ROOTS_GUI_UI_SOURCE}")
  message(FATAL_ERROR "Prepare verified local sources and set ROOTS_GUI_UI_SOURCE")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ui-dependencies.cmake")
roots_gui_add_ui_dependencies("${ROOTS_GUI_UI_SOURCE}")
add_executable(roots_gui_imkit_consumer consumer.cpp)
target_link_libraries(roots_gui_imkit_consumer PRIVATE imkit::imkit roots_gui_backends)
target_compile_features(roots_gui_imkit_consumer PRIVATE cxx_std_20)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/service-dependencies.cmake")
set(BUILD_SHARED_LIBS ON)
set(BUILD_TESTING ON)
roots_gui_add_service_dependencies("${ROOTS_GUI_UI_SOURCE}")
if(NOT BUILD_SHARED_LIBS OR NOT BUILD_TESTING)
  message(FATAL_ERROR "GUI service dependencies changed caller build options")
endif()
add_executable(roots_gui_service_consumer service_consumer.cpp)
target_link_libraries(roots_gui_service_consumer PRIVATE roots_gui_http roots_gui_sqlite)
target_compile_features(roots_gui_service_consumer PRIVATE cxx_std_20)
enable_testing()
add_test(NAME roots_gui.imkit_consumer COMMAND roots_gui_imkit_consumer "${ROOTS_GUI_UI_SOURCE}/imkit/assets/fonts/Inter-Regular.ttf")
add_test(NAME roots_gui.service_consumer COMMAND roots_gui_service_consumer)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(roots_gui_native_consumer native_consumer.cpp)
  target_link_libraries(roots_gui_native_consumer PRIVATE imkit::imkit roots_gui_backends)
  target_compile_features(roots_gui_native_consumer PRIVATE cxx_std_20)
  # Run explicitly under an owned display; not part of the socket-denied CPU suite.
endif()
