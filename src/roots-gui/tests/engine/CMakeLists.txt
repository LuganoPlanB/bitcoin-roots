  "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/prepare_test_inputs.py" --destination "${test_inputs}"
  RESULT_VARIABLE prepared OUTPUT_VARIABLE preparation ERROR_VARIABLE preparation_error)

  message(FATAL_ERROR "Local test source preparation failed: ${preparation_error}")
endif()

  "${test_inputs}" RESULT_VARIABLE verified OUTPUT_VARIABLE verification)

  message(FATAL_ERROR "Test input verification failed: ${verification}")
