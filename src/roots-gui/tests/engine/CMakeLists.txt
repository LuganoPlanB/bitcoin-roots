# Test-only copies: no production ImGui objects or hook-bearing objects are shared.
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/test-inputs-lock.json" test_lock)
string(JSON test_archive_sha GET "${test_lock}" sha256)
set(test_inputs "${CMAKE_CURRENT_BINARY_DIR}/inputs/test-${test_archive_sha}")
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/prepare_test_inputs.py" --destination "${test_inputs}"
  RESULT_VARIABLE prepared OUTPUT_VARIABLE preparation ERROR_VARIABLE preparation_error)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Local test source preparation failed: ${preparation_error}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/input-preparation.json" "${preparation}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_inputs.py"
  "${test_inputs}" RESULT_VARIABLE verified OUTPUT_VARIABLE verification)
if(NOT verified EQUAL 0)
  message(FATAL_ERROR "Test input verification failed: ${verification}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/input-verification.json" "${verification}")
add_library(roots_gui_test_config INTERFACE)
target_compile_features(roots_gui_test_config INTERFACE cxx_std_20)
target_compile_definitions(roots_gui_test_config INTERFACE IMGUI_USER_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config.h")
if(MSVC)
  target_compile_options(roots_gui_test_config INTERFACE /UNDEBUG)
else()
  target_compile_options(roots_gui_test_config INTERFACE -UNDEBUG)
endif()
add_library(roots_gui_test_imgui STATIC
  "${roots_gui_ui_source}/imgui/imgui.cpp" "${roots_gui_ui_source}/imgui/imgui_draw.cpp"
  "${roots_gui_ui_source}/imgui/imgui_tables.cpp" "${roots_gui_ui_source}/imgui/imgui_widgets.cpp"
  "${test_inputs}/imgui/backends/imgui_impl_null.cpp")
target_include_directories(roots_gui_test_imgui PUBLIC "${roots_gui_ui_source}/imgui" "${test_inputs}/imgui/backends")
target_link_libraries(roots_gui_test_imgui PUBLIC roots_gui_test_config)
set(engine "${test_inputs}/engine/imgui_test_engine")
add_library(roots_gui_test_engine STATIC)
foreach(unit imgui_capture_tool imgui_te_context imgui_te_coroutine imgui_te_engine imgui_te_exporters imgui_te_perftool imgui_te_ui imgui_te_utils)
  target_sources(roots_gui_test_engine PRIVATE "${engine}/${unit}.cpp")
endforeach()
target_include_directories(roots_gui_test_engine PUBLIC "${test_inputs}/engine")
target_link_libraries(roots_gui_test_engine PUBLIC roots_gui_test_imgui Threads::Threads)
# Core hooks call the engine, whose implementation calls core. CMake repeats
# this test-only static-library component to resolve both directions.
target_link_libraries(roots_gui_test_imgui PRIVATE roots_gui_test_engine)
add_library(roots_gui_test_imkit STATIC)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/ui-lock.json" ui_lock)
string(JSON count LENGTH "${ui_lock}" imkit coreSources)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON unit GET "${ui_lock}" imkit coreSources ${index})
  target_sources(roots_gui_test_imkit PRIVATE "${roots_gui_ui_source}/imkit/src/${unit}.cpp")
endforeach()
target_include_directories(roots_gui_test_imkit PUBLIC "${roots_gui_ui_source}/imkit/include")
target_link_libraries(roots_gui_test_imkit PUBLIC roots_gui_test_imgui)
add_executable(roots_gui_engine_tests runner.cpp tests.cpp ../../ui/bootstrap.cpp ../../ui/preferences.cpp ../../application/preferences.cpp)
target_include_directories(roots_gui_engine_tests PRIVATE ../..)
target_link_libraries(roots_gui_engine_tests PRIVATE roots_gui_test_engine roots_gui_test_imkit)
add_executable(roots_gui_preference_models preferences_model.cpp ../../application/preferences.cpp)
target_include_directories(roots_gui_preference_models PRIVATE ../..)
target_compile_features(roots_gui_preference_models PRIVATE cxx_std_20)
add_executable(roots_gui_models ../model/model_tests.cpp ../../domain/money.cpp ../../domain/workflow.cpp)
target_include_directories(roots_gui_models PRIVATE ../..)
target_compile_features(roots_gui_models PRIVATE cxx_std_20)
foreach(kind theme design_system workflow charts api_compile design_api_compile charts_api_compile)
  if(kind STREQUAL "workflow")
    add_executable(roots_gui_imkit_${kind} workflow_core.cpp)
  else()
    add_executable(roots_gui_imkit_${kind} "${test_inputs}/imkit/tests/${kind}.cpp")
  endif()
  if(kind STREQUAL "design_system")
    target_sources(roots_gui_imkit_${kind} PRIVATE "${test_inputs}/imkit/examples/gallery/allocation_probe.cpp")
  endif()
  target_link_libraries(roots_gui_imkit_${kind} PRIVATE roots_gui_test_imkit roots_gui_test_engine)
  add_test(NAME roots_gui.imkit.${kind} COMMAND roots_gui_imkit_${kind})
  set_tests_properties(roots_gui.imkit.${kind} PROPERTIES TIMEOUT 30 LABELS "roots-gui;imkit;cpu;ui;roots-gui.ui")
endforeach()
foreach(kind preferences bootstrap)
  add_test(NAME roots_gui.engine.${kind} COMMAND roots_gui_engine_tests "roots.${kind}" "${CMAKE_CURRENT_BINARY_DIR}/${kind}.xml" 1)
  set_tests_properties(roots_gui.engine.${kind} PROPERTIES TIMEOUT 30 LABELS "roots-gui;engine;interaction;ui;roots-gui.ui")
endforeach()
foreach(probe failure unknown count undrained timeout)
  add_test(NAME roots_gui.runner.${probe} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_runner.py"
    --runner "$<TARGET_FILE:roots_gui_engine_tests>" --probe "${probe}" --output "${CMAKE_CURRENT_BINARY_DIR}/${probe}")
  set_tests_properties(roots_gui.runner.${probe} PROPERTIES TIMEOUT 10 LABELS "roots-gui;runner;negative;boundary;roots-gui.boundary")
endforeach()
add_test(NAME roots_gui.models COMMAND roots_gui_models)
add_test(NAME roots_gui.preferences.models COMMAND roots_gui_preference_models)
foreach(kind fixture_contracts screen_contracts)
  add_test(NAME roots_gui.${kind} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../model/${kind}.py")
  set_tests_properties(roots_gui.${kind} PROPERTIES TIMEOUT 30 LABELS "roots-gui;model;contracts;roots-gui.model")
endforeach()
set_tests_properties(roots_gui.models roots_gui.preferences.models PROPERTIES TIMEOUT 30 LABELS "roots-gui;model;roots-gui.model")
